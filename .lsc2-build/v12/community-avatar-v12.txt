function communityInitials(value:string) {
  return value.split(/\s+/).filter(Boolean).slice(0,2).map((part)=>part[0]?.toUpperCase()).join('') || 'LS'
}
function CommunityAvatarContent({name,src}:{name:string;src?:string|null}) {
  return <>
    {src ? <img src={src} alt="" onError={(event) => event.currentTarget.remove()} /> : null}
    <span className="avatar-fallback">{communityInitials(name)}</span>
  </>
}
