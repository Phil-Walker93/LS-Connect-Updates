function sameMessageDay(a:string,b:string){return new Date(a).toDateString()===new Date(b).toDateString()}
function messageDateLabel(v:string){const d=new Date(v),t=new Date(),y=new Date(t);y.setDate(t.getDate()-1);if(d.toDateString()===t.toDateString())return'Heute';if(d.toDateString()===y.toDateString())return'Gestern';return d.toLocaleDateString('de-DE',{day:'2-digit',month:'long',year:'numeric'})}
function fullMessageDate(v:string){return new Date(v).toLocaleString('de-DE',{day:'2-digit',month:'2-digit',year:'numeric',hour:'2-digit',minute:'2-digit',second:'2-digit'})}
function relativePresence(v:string|null){if(!v)return'Offline';const m=Math.floor(Math.max(0,Date.now()-new Date(v).getTime())/60000);if(m<1)return'Gerade eben online';if(m<60)return'Zuletzt online vor '+m+' Min.';const h=Math.floor(m/60);if(h<24)return'Zuletzt online vor '+h+' Std.';return'Zuletzt online am '+new Date(v).toLocaleDateString('de-DE')}
function friendlyMessengerError(reason:unknown,fallback:string){const raw=reason instanceof Error?reason.message:String(reason??'');if(raw.includes('recipient does not accept messages'))return'Nachricht konnte nicht gesendet werden. Dieser Nutzer akzeptiert derzeit keine Nachrichten von dir.';if(raw.includes('contact required'))return'Dieser Nutzer akzeptiert nur Nachrichten von Kontakten.';if(raw.includes('conversation unavailable'))return'Dieser Chat ist nicht mehr verfügbar.';if(raw.includes('profile unavailable'))return'Dieser Charakter ist nicht mehr verfügbar.';return raw||fallback}
